## 1. 故障排除概述

### 故障排除原则

1. **先检查日志**：查看详细的错误信息
2. **隔离问题**：确定是哪个模块的问题

- **配置错误**：环境变量、配置文件问题

- **AI错误**：API响应、解析问题

```python
# 在src/analyzer.py中添加重试机制
import time
from functools import wraps

def retry_on_rate_limit(max_retries=3, delay=1):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except RateLimitError:
                    if attempt == max_retries - 1:
                        raise
                    time.sleep(delay * (attempt + 1))
            return wrapper
        return wrapper
    return decorator
```

**预防措施：**
- 实现指数退避重试
- 添加请求间隔
- 使用缓存减少重复请求

```python
    path = Path(file_path)
```

```python
# 增强JSON解析
import re
import json

def parse_ai_response(response_text):
    """增强的AI响应解析"""
    # 尝试直接解析
    try:
        return json.loads(response_text)
    except json.JSONDecodeError:
        pass
    
    # 尝试提取JSON部分
    json_patterns = [
        r'\{[^{}]*\}',  # 简单JSON对象
        r'\{.*\}',      # 复杂JSON对象
        r'\[.*\]',      # JSON数组
    ]
    
    for pattern in json_patterns:
        matches = re.findall(pattern, response_text, re.DOTALL)
        for match in matches:
            try:
                return json.loads(match)
            except json.JSONDecodeError:
                continue
    
    # 返回默认结构
    return {
        "characters": [],
        "settings": [],
        "events": [],
        "summary": "AI返回格式错误，请重试"
    }
```

```python
# 检查wikilink格式
def validate_wikilink(link):
    """验证wikilink格式"""
    if not link.startswith('[[') or not link.endswith(']]'):
        return False
    
    # 检查是否有无效字符
    content = link[2:-2]
    invalid_chars = ['/', '\\', ':', '*', '?', '"', '<', '>', '|']
    for char in invalid_chars:
        if char in content:
            return False
    
    return True

# 修复wikilink
def fix_wikilink(link):
    """修复wikilink格式"""
    content = link[2:-2]
    
    # 移除无效字符
    invalid_chars = ['/', '\\', ':', '*', '?', '"', '<', '>', '|']
    for char in invalid_chars:
        content = content.replace(char, '_')
    
    return f"[[{content}]]"
```

### 6.3 文件命名问题


**解决方案：**
```python
import re
import os

def sanitize_filename(filename):
    """清理文件名中的非法字符"""
    # 移除非法字符
    illegal_chars = r'[<>:"/\\|?*]'
    clean_name = re.sub(illegal_chars, '_', filename)
    
    # 限制长度
    if len(clean_name) > 255:
        name, ext = os.path.splitext(clean_name)
        clean_name = name[:255-len(ext)] + ext
    
    return clean_name
```

```python
def check_chapter_duplicates(new_chapter, existing_chapters):
    """检查章节重复"""
    from difflib import SequenceMatcher
    
    for i, chapter in enumerate(existing_chapters):
        similarity = SequenceMatcher(None, 
                                   new_chapter['content'], 
                                   chapter['content']).ratio()
        
        if similarity > 0.7:  # 相似度超过70%
            return True, f"与第{i+1}章相似度过高: {similarity:.2%}"
```

```python
        backupCount=3
    )
    error_handler.setLevel(logging.ERROR)
    
    # 格式化器
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    )
    
    file_handler.setFormatter(formatter)
    error_handler.setFormatter(formatter)
    
    logger.addHandler(file_handler)
    logger.addHandler(error_handler)
    
    return logger
```
